# Rol Recursos Humanos (RR.HH) — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Agregar el rol `rrhh` con funcionalidades de asesor + un Panel RR.HH propio con páginas Comunidad Altaltium (directorio de empleados) y Organigrama (importado desde Excel), eliminando el organigrama viejo.

**Architecture:** Nuevo módulo MVC espejo del patrón admin: rutas `/rrhh/*` protegidas con `requireRole('rrhh','directivo','admin')`, controller propio, vistas EJS con layout `main.ejs` y sidebar partial propio que reutiliza los IDs `admin-sidebar`/`admin-overlay`/`admin-hamburger` para heredar el drawer móvil de `drawer.js` y el `margin-left` global de `styles.css`. El organigrama se guarda en la tabla nueva `organigrama_empleados` (reemplazo completo por importación) y las tablas viejas `org_*` se eliminan.

**Tech Stack:** Node.js 24 + Express 5, EJS + express-ejs-layouts, PostgreSQL (pg), multer (memoria) + xlsx para importar, Tailwind CDN + estilos teal del proyecto.

**Spec:** `docs/superpowers/specs/2026-07-16-rol-rrhh-design.md`

**Notas de ejecución (reglas del proyecto):**
- No hay framework de tests en el proyecto — la verificación de cada tarea es manual en localhost con `npm run dev` (flujo local-primero del usuario).
- **Commits:** solo con autorización del usuario. Si no está autorizado aún, salta los pasos de commit y déjalos pendientes al final.
- La migración SQL se corre primero en la BD local; el usuario la aplica en Render (DATABASE_URL) antes del deploy.
- El CSRF propio del proyecto exige token en cada POST. En formularios multipart el token va en el **query string** del action: `?_csrf=<%= encodeURIComponent(csrfToken) %>` (patrón de `views/admin/inventario-outlet/importar.ejs:188`).

// src/controllers/rrhh/panel.controller.js
'use strict';

const pool = require('../../db/pool');

const ROLE_LABELS = {
  admin:          'Administrador',
  manager:        'Gerente',
  directivo:      'Directivo',
  marketing:      'Marketing',
  administracion: 'Administración',
  externo:        'Externo',
  advisor:        'Asesor Comercial',
  rrhh:           'Recursos Humanos',
};

// Normaliza nombres para cruzar empleado ↔ jefe (acentos, espacios, mayúsculas)
function normalizarNombre(s) {
  return String(s || '')
    .normalize('NFD')
    .replace(/\p{M}/gu, '')
    .replace(/\s+/g, ' ')
    .trim()
    .toLowerCase();
}

// Construye el árbol jerárquico. Raíces = sin jefe o jefe no encontrado.
// Protegido contra ciclos (A jefe de B y B jefe de A).
function construirArbol(empleados) {
  const nodos = empleados.map(e => ({ ...e, hijos: [] }));
  const porNombre = new Map();
  nodos.forEach(n => porNombre.set(normalizarNombre(n.nombre), n));

  const raices = [];
  nodos.forEach(n => {
    const jefe = n.jefe_nombre ? porNombre.get(normalizarNombre(n.jefe_nombre)) : null;
    if (jefe && jefe !== n) jefe.hijos.push(n);
    else raices.push(n);
  });

  const visitados = new Set();
  function marcar(n) {
    if (visitados.has(n.id)) return;
    visitados.add(n.id);
    n.hijos.forEach(marcar);
  }
  raices.forEach(marcar);

  // Nodos en ciclo: se rompe el ciclo y suben a raíz
  nodos.forEach(n => {
    if (!visitados.has(n.id)) {
      const jefe = porNombre.get(normalizarNombre(n.jefe_nombre));
      if (jefe) jefe.hijos = jefe.hijos.filter(h => h.id !== n.id);
      raices.push(n);
      marcar(n);
    }
  });

  return raices;
}

// GET /rrhh
exports.index = (req, res) => res.redirect('/rrhh/comunidad');

// GET /rrhh/comunidad — directorio de empleados activos agrupado por área
exports.comunidad = async (req, res) => {
  try {
    const { rows: empleados } = await pool.query(`
      SELECT id, nombre, apellidos, email, telefono, role, area, avatar_url
      FROM users
      WHERE is_active = true
      ORDER BY COALESCE(NULLIF(TRIM(area), ''), 'zzz') ASC, nombre ASC, apellidos ASC
    `);

    const grupos = new Map();
    empleados.forEach(e => {
      const key = (e.area && e.area.trim()) ? e.area.trim() : 'General';
      if (!grupos.has(key)) grupos.set(key, []);
      grupos.get(key).push(e);
    });

    return res.render('rrhh/comunidad', {
      title: 'Comunidad Altaltium | RR.HH',
      user: req.session.user,
      grupos: [...grupos.entries()], // [[area, [empleados]], ...]
      totalEmpleados: empleados.length,
      roleLabels: ROLE_LABELS,
      path: '/rrhh/comunidad',
    });
  } catch (err) {
    console.error('[rrhh] comunidad:', err.message);
    return res.status(500).send('Error al cargar la comunidad');
  }
};

// GET /rrhh/organigrama
exports.organigrama = async (req, res) => {
  try {
    const { rows } = await pool.query(`
      SELECT id, nombre, puesto, area, jefe_nombre, orden
      FROM organigrama_empleados
      ORDER BY orden ASC, id ASC
    `);
    return res.render('rrhh/organigrama', {
      title: 'Organigrama | RR.HH',
      user: req.session.user,
      raices: construirArbol(rows),
      totalEmpleados: rows.length,
      puedeImportar: ['rrhh', 'admin'].includes(req.session.user.role),
      path: '/rrhh/organigrama',
    });
  } catch (err) {
    console.error('[rrhh] organigrama:', err.message);
    return res.status(500).send('Error al cargar el organigrama');
  }
};

// GET /rrhh/organigrama/importar
exports.importarForm = (req, res) => {
  return res.render('rrhh/organigrama-importar', {
    title: 'Importar Organigrama | RR.HH',
    user: req.session.user,
    resultado: null,
    path: '/rrhh/organigrama/importar',

};

// POST /rrhh/organigrama/importar — reemplaza el organigrama completo
exports.importarProcesar = async (req, res) => {
  const renderResultado = (resultado) => res.render('rrhh/organigrama-importar', {
    title: 'Importar Organigrama | RR.HH',
    user: req.session.user,
    resultado,
    path: '/rrhh/organigrama/importar',
  });
  const renderError = (motivo) =>
    renderResultado({ importados: 0, ignoradas: [], jefesNoEncontrados: [], error: motivo });

  if (!req.file) return renderError('No se seleccionó ningún archivo.');

  let rows;
  try {
    const XLSX = require('xlsx');
    const wb = XLSX.read(req.file.buffer, { type: 'buffer' });
    const ws = wb.Sheets[wb.SheetNames[0]];
    rows = XLSX.utils.sheet_to_json(ws, { defval: '' });
  } catch (err) {
    console.error('[rrhh] importar leer:', err.message);
    return renderError('No se pudo leer el archivo. Verifica que sea un .xlsx, .xls o .csv válido.');
  }
  if (!rows.length) return renderError('El archivo está vacío o no tiene filas de datos.');

  // Encabezados tolerantes a mayúsculas y acentos
  const buscarCampo = (fila, claves) => {
    for (const k of Object.keys(fila)) {
      if (claves.includes(normalizarNombre(k))) return String(fila[k]).trim();
    }
    return '';
  };

  const empleados = [];
  const ignoradas = [];
  rows.forEach((fila, i) => {
    const nombre = buscarCampo(fila, ['nombre', 'nombre completo', 'empleado']);
    if (!nombre) {
      ignoradas.push({ fila: i + 2, motivo: 'Sin nombre' }); // +2: encabezado + índice base 1
      return;
    }
    empleados.push({
      nombre,
      puesto: buscarCampo(fila, ['puesto', 'cargo']),
      area:   buscarCampo(fila, ['area', 'departamento']),   // 'área' → 'area' al normalizar
      jefe:   buscarCampo(fila, ['jefe', 'jefe directo', 'reporta a', 'reporta']),
      orden:  i,
    });
  });

  if (!empleados.length) {
    return renderError('Ninguna fila tiene la columna "Nombre" con datos. Revisa los encabezados (Nombre | Puesto | Área | Jefe).');
  }

  const nombres = new Set(empleados.map(e => normalizarNombre(e.nombre)));
  const jefesNoEncontrados = [...new Set(
    empleados
      .filter(e => e.jefe && !nombres.has(normalizarNombre(e.jefe)))
      .map(e => e.jefe)
  )];

  const client = await pool.connect();
  try {
    await client.query('BEGIN');
    await client.query('DELETE FROM organigrama_empleados');
    for (const e of empleados) {
      await client.query(
        `INSERT INTO organigrama_empleados (nombre, puesto, area, jefe_nombre, orden)
         VALUES ($1, NULLIF($2,''), NULLIF($3,''), NULLIF($4,''), $5)`,
        [e.nombre, e.puesto, e.area, e.jefe, e.orden]
      );
    }
    await client.query('COMMIT');
  } catch (err) {
    await client.query('ROLLBACK');
    console.error('[rrhh] importar BD:', err.message);
    return renderError('Error al guardar en la base de datos. El organigrama anterior quedó intacto.');
  } finally {
    client.release();
  }

  return renderResultado({
    importados: empleados.length,
    ignoradas,
    jefesNoEncontrados,
    error: null,
  });
};

// GET /rrhh/organigrama/plantilla — Excel de ejemplo
exports.descargarPlantilla = (req, res) => {
  const XLSX = require('xlsx');
  const datos = [
    { Nombre: 'María López',  Puesto: 'Directora General', 'Área': 'Dirección', Jefe: '' },
    { Nombre: 'Juan Pérez',   Puesto: 'Gerente Comercial', 'Área': 'Ventas',    Jefe: 'María López' },
    { Nombre: 'Ana Torres',   Puesto: 'Asesora',           'Área': 'Ventas',    Jefe: 'Juan Pérez' },
  ];
  const ws = XLSX.utils.json_to_sheet(datos);
  const wb = XLSX.utils.book_new();
  XLSX.utils.book_append_sheet(wb, ws, 'Organigrama');
  const buf = XLSX.write(wb, { type: 'buffer', bookType: 'xlsx' });
  res.setHeader('Content-Disposition', 'attachment; filename="plantilla_organigrama.xlsx"');
  res.setHeader('Content-Type', 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet');
  return res.send(buf);
};
```

// src/routes/rrhh.routes.js
'use strict';

const express = require('express');
const router  = express.Router();
const multer  = require('multer');

const { ensureAuth, requireRole } = require('../middlewares/auth');
const ctrl = require('../controllers/rrhh/panel.controller');

const uploadExcel = multer({
  storage: multer.memoryStorage(),
  limits: { fileSize: 10 * 1024 * 1024 },
  fileFilter: (req, file, cb) => {
    const ok = /\.(xlsx|xls|csv)$/i.test(file.originalname || '');
    if (ok) return cb(null, true);
    return cb(new Error('Formato no soportado. Usa .xlsx, .xls o .csv'));
  },
});

// requireRole('rrhh','directivo','admin'): el nivel máximo listado es admin (5),
// así que manager/advisor/marketing NO pasan por jerarquía — solo match directo.
const verPanel  = [ensureAuth, requireRole('rrhh', 'directivo', 'admin')];
const gestionar = [ensureAuth, requireRole('rrhh', 'admin')];

router.get('/rrhh',                        verPanel,  ctrl.index);
router.get('/rrhh/comunidad',              verPanel,  ctrl.comunidad);
router.get('/rrhh/organigrama',            verPanel,  ctrl.organigrama);
router.get('/rrhh/organigrama/importar',   gestionar, ctrl.importarForm);
router.post('/rrhh/organigrama/importar',  gestionar, uploadExcel.single('archivo'), ctrl.importarProcesar);
router.get('/rrhh/organigrama/plantilla',  gestionar, ctrl.descargarPlantilla);

module.exports = router;
```
